    function calcAgeGroup(dateString) {
      if (!dateString) return "未入力";
      const year = Number(dateString.slice(0, 4));
      if (!year) return "未入力";
      const now = new Date().getFullYear();
      const age = now - year;
      if (age < 20) return "10代以下";
      if (age < 30) return "20代";
      if (age < 40) return "30代";
      if (age < 50) return "40代";
      if (age < 60) return "50代";
      return "60代以上";
    }

    function luckByBirth(dateString) {
      if (!dateString) return "大吉";
      const nums = dateString.replaceAll("-", "").split("").map(Number);
      const sum = nums.reduce((a, b) => a + b, 0);
      return fortunes[sum % fortunes.length];
    }

    function luckByText(text) {
      if (!text) return "開運準備期";
      const sum = Array.from(text).reduce((acc, char) => acc + char.charCodeAt(0), 0);
      return fortunes[sum % fortunes.length];
    }
